// Implementation generated to pass all tests
use axum::{
    extract::State,
    http::StatusCode,
    response::Json,
    routing::post,
    Router,
};
use serde::{Deserialize, Serialize};
use std::collections::HashMap;
use std::sync::Arc;
use std::time::{Duration, Instant};

#[derive(Debug, Deserialize)]
pub struct LoginRequest {
    pub email: String,
    pub password: String,
    #[serde(default)]
    pub remember_me: bool,
}

#[derive(Debug, Serialize)]
pub struct LoginResponse {
    pub token: String,
    pub user: UserData,
    pub expires_in: u64,
}

#[derive(Debug, Deserialize)]
pub struct LoginErrorResponse {
    pub error: String,
    pub code: String,
}

#[derive(Clone)]
struct RateLimitStore {
    attempts: Arc<std::sync::RwLock<HashMap<String, (u32, Instant)>>>,
}

impl RateLimitStore {
    fn new() -> Self {
        Self {
            attempts: Arc::new(std::sync::RwLock::new(HashMap::new())),
        }
    }

    fn check_rate_limit(&self, key: &str) -> bool {
        let mut attempts = self.attempts.write().unwrap();
        let now = Instant::now();

        match attempts.get_mut(key) {
            Some((count, last_attempt)) => {
                if now.duration_since(*last_attempt) > Duration::from_secs(60) {
                    *count = 1;
                    *last_attempt = now;
                    true
                } else if *count >= 5 {
                    false
                } else {
                    *count += 1;
                    true
                }
            }
            None => {
                attempts.insert(key.to_string(), (1, now));
                true
            }
        }
    }
}

pub async fn login(
    State(rate_limit): State<RateLimitStore>,
    Json(request): Json<LoginRequest>,
) -> Result<Json<LoginResponse>, (StatusCode, Json<LoginErrorResponse>)> {
    // Validate request
    if request.email.is_empty() || request.password.is_empty() {
        return Err((
            StatusCode::BAD_REQUEST,
            Json(LoginErrorResponse {
                error: "Email and password are required".to_string(),
                code: "MISSING_FIELDS".to_string(),
            }),
        ));
    }

    // Rate limiting
    let rate_limit_key = format!("login:{}", request.email);
    if !rate_limit.check_rate_limit(&rate_limit_key) {
        return Err((
            StatusCode::TOO_MANY_REQUESTS,
            Json(LoginErrorResponse {
                error: "Too many login attempts. Please try again later.".to_string(),
                code: "RATE_LIMITED".to_string(),
            }),
        ));
    }

    // Authenticate user
    let auth_service = AuthService::new();
    match auth_service.login(request).await {
        Ok(response) => {
            // Log successful login
            tracing::info!("User logged in: {}", response.user.email);

            Ok(Json(response))
        }
        Err(e) => {
            // Log failed login
            tracing::warn!("Failed login attempt for: {}", request.email);

            let (status, code) = match e {
                AuthError::InvalidCredentials => (StatusCode::UNAUTHORIZED, "INVALID_CREDENTIALS"),
                AuthError::UserNotFound => (StatusCode::UNAUTHORIZED, "USER_NOT_FOUND"),
                AuthError::InternalError(_) => (StatusCode::INTERNAL_SERVER_ERROR, "INTERNAL_ERROR"),
            };

            Err((
                status,
                Json(LoginErrorResponse {
                    error: e.to_string(),
                    code: code.to_string(),
                }),
            ))
        }
    }
}

pub fn create_auth_router() -> Router {
    Router::new()
        .route("/login", post(login))
        .with_state(RateLimitStore::new())
}
